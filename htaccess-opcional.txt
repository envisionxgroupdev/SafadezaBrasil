# ---- Safadeza Brasil — regras OPCIONAIS ----
# O site funciona sem este arquivo. NÃO substitua o .htaccess da sua hospedagem por ele:
# lá ficam as linhas que ligam o PHP e o HTTPS.
# Se quiser os "links bonitos" (PRETTY_URLS) ou /sitemap.xml, COPIE o bloco abaixo e
# COLE NO FINAL do .htaccess que já existe na pasta principal do site.

<IfModule mod_rewrite.c>
  RewriteEngine On
  # Endereços de pasta abrem o index.php
  RewriteRule ^$ index.php [L]
  RewriteRule ^admin/?$ admin/index.php [L]
  # Bloqueia pastas e arquivos internos
  RewriteRule ^(data|install|includes)(/|$) - [F,L]
  RewriteRule \.(sql|json|lock|md|log)$ - [F,L]
  RewriteRule (^|/)\.(?!well-known) - [F,L]
  # Links bonitos (ative PRETTY_URLS no config.php)
  RewriteRule ^video/([0-9]+)(?:/([^/]*))?/?$ video.php?id=$1&s=$2 [L,QSA]
  RewriteRule ^categoria/([^/]+)/?$ category.php?c=$1 [L,QSA]
  # SEO: robots.txt e sitemap.xml
  RewriteCond %{REQUEST_FILENAME} !-f
  RewriteRule ^robots\.txt$ robots.php [L]
  RewriteRule ^sitemap\.xml$ sitemap.php [L]
</IfModule>

<IfModule mod_headers.c>
  Header always set X-Content-Type-Options "nosniff"
  Header always set Rating "RTA-5042-1996-1400-1577-RTA"
</IfModule>
